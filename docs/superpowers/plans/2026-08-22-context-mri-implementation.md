# Context MRI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the four-component Context MRI MVP (SDK, backend, demo app, dashboard) so it can run the insurance-claims healthy/failure demo end to end.

**Architecture:** A Python SDK wraps the demo app's LLM calls, labeling every piece of context as a `Segment`, rendering within a token budget, and diffing each call's segments against the previous call in the same trace. Every call posts an event to an Express + SQLite backend, which stores it and raises an alert when a segment marked `critical` disappears between two consecutive calls. A React dashboard reads the backend and shows context composition, before/after diffs, and a timeline that links an eviction straight to the tool call that followed it.

**Tech Stack:** Node.js 22 / Express 5 / `node:sqlite` (backend), Python 3.12 / `requests` / `pytest` (SDK + demo app), Vite + React 18 (dashboard), OpenAI Responses API (existing `context_check.py` helper).

**Design correction found during planning:** the architecture doc's description of `render()` ("drop lowest-priority segments first, critical segments survive longest") would make the critical policy segment **unevictable** in practice, because sorting critical-first means it's always the first segment considered and always fits. That silently breaks the entire failure demo — the policy would never disappear no matter how much history piles up. The plan below instead has `render()` do **naive recency-based truncation**: keep the most recently added segments, and drop from the *oldest*-added segment backward once the budget is exceeded. Since the policy segment is always the first thing added each turn, it becomes the first candidate dropped once volume is high enough — which is also a more realistic bug (naive sliding-window truncation dropping the system prompt is a well-documented real failure mode). The `critical` flag is no longer involved in eviction order at all; it's now purely a label the backend uses for alerting. Task 7's tests reflect this corrected behavior. `architecture-idea.md` should be updated to match — flagged for you to review after this plan.

const { DatabaseSync } = require('node:sqlite');
const path = require('node:path');

const DEFAULT_DB_PATH = path.join(__dirname, 'context_mri.db');

const SCHEMA = `
CREATE TABLE IF NOT EXISTS traces (
  trace_id TEXT PRIMARY KEY,
  created_at TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS model_calls (
  call_id TEXT PRIMARY KEY,
  trace_id TEXT NOT NULL,
  timestamp TEXT NOT NULL,
  token_budget INTEGER NOT NULL,
  token_used INTEGER NOT NULL,
  model_response_json TEXT
);

CREATE TABLE IF NOT EXISTS context_segments (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  call_id TEXT NOT NULL,
  segment_id TEXT NOT NULL,
  source TEXT NOT NULL,
  critical INTEGER NOT NULL,
  token_count INTEGER NOT NULL,
  present INTEGER NOT NULL
);

CREATE TABLE IF NOT EXISTS lifecycle_events (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  call_id TEXT NOT NULL,
  segment_id TEXT NOT NULL,
  type TEXT NOT NULL,
  previous_call_id TEXT
);

CREATE TABLE IF NOT EXISTS tool_calls (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  call_id TEXT NOT NULL,
  tool_name TEXT NOT NULL,
  args_json TEXT NOT NULL,
  result_json TEXT
);

CREATE TABLE IF NOT EXISTS alerts (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  trace_id TEXT NOT NULL,
  call_id TEXT NOT NULL,
  previous_call_id TEXT NOT NULL,
  segment_id TEXT NOT NULL,
  created_at TEXT NOT NULL
);
`;

function createDb(dbPath = process.env.DB_PATH || DEFAULT_DB_PATH) {
  const db = new DatabaseSync(dbPath);
  db.exec(SCHEMA);
  return db;
}

module.exports = { createDb, DEFAULT_DB_PATH };
```

// backend/lib/alertRules.js
function computeAlerts({ events, previousSegmentsById }) {
  const alerts = [];
  for (const event of events) {
    if (event.type !== 'evicted') continue;
    const previousSegment = previousSegmentsById.get(event.segment_id);
    if (previousSegment && previousSegment.critical) {
      alerts.push({ segment_id: event.segment_id, previous_call_id: event.previous_call_id });
    }
  }
  return alerts;
}

module.exports = { computeAlerts };
```

// backend/controllers/eventsController.js
const { computeAlerts } = require('../lib/alertRules');

function createEventsController(db) {
  const insertTrace = db.prepare('INSERT OR IGNORE INTO traces (trace_id, created_at) VALUES (?, ?)');
  const insertCall = db.prepare(`
    INSERT INTO model_calls (call_id, trace_id, timestamp, token_budget, token_used, model_response_json)
    VALUES (?, ?, ?, ?, ?, ?)
  `);
  const insertSegment = db.prepare(`
    INSERT INTO context_segments (call_id, segment_id, source, critical, token_count, present)
    VALUES (?, ?, ?, ?, ?, ?)
  `);
  const insertEvent = db.prepare(`
    INSERT INTO lifecycle_events (call_id, segment_id, type, previous_call_id)
    VALUES (?, ?, ?, ?)
  `);
  const insertAlert = db.prepare(`
    INSERT INTO alerts (trace_id, call_id, previous_call_id, segment_id, created_at)
    VALUES (?, ?, ?, ?, ?)
  `);
  const segmentsForCall = db.prepare('SELECT segment_id, critical FROM context_segments WHERE call_id = ?');

  return function handleCreateEvent(req, res) {
    const {
      trace_id: traceId,
      call_id: callId,
      timestamp,
      segments = [],
      events = [],
      token_budget: tokenBudget,
      token_used: tokenUsed,
      model_response: modelResponse,
    } = req.body;

    if (!traceId || !callId || !timestamp) {
      res.status(400).json({ error: 'trace_id, call_id, and timestamp are required' });
      return;
    }

    insertTrace.run(traceId, timestamp);
    insertCall.run(callId, traceId, timestamp, tokenBudget ?? 0, tokenUsed ?? 0, JSON.stringify(modelResponse ?? null));

    for (const segment of segments) {
      insertSegment.run(
        callId,
        segment.segment_id,
        segment.source,
        segment.critical ? 1 : 0,
        segment.token_count,
        segment.present ? 1 : 0,
      );
    }

    for (const event of events) {
      insertEvent.run(callId, event.segment_id, event.type, event.previous_call_id ?? null);
    }

    const previousCallId = events.find((e) => e.previous_call_id)?.previous_call_id;
    if (previousCallId) {
      const previousSegments = segmentsForCall.all(previousCallId);
      const previousSegmentsById = new Map(
        previousSegments.map((s) => [s.segment_id, { critical: !!s.critical }]),
      );
      const alerts = computeAlerts({ events, previousSegmentsById });
      for (const alert of alerts) {
        insertAlert.run(traceId, callId, alert.previous_call_id, alert.segment_id, new Date().toISOString());
      }
    }

    res.status(201).json({ ok: true });
  };
}

module.exports = { createEventsController };
```

# sdk/context_mri/builder.py
from __future__ import annotations

import uuid
from dataclasses import dataclass, field

from .models import Segment


@dataclass
class ContextBuilder:
    trace_id: str
    call_id: str = field(default_factory=lambda: str(uuid.uuid4()))
    segments: list[Segment] = field(default_factory=list)
    _rendered: list[Segment] = field(default_factory=list, init=False, repr=False)

    def add(self, source: str, content: str, *, critical: bool = False, priority: int = 0) -> "ContextBuilder":
        self.segments.append(Segment(source=source, content=content, critical=critical, priority=priority))
        return self

    def render(self, max_tokens: int) -> list[Segment]:
        """Naive recency-based truncation: keep the most recently added
        segments first, dropping from the oldest-added segment backward
        once the budget is exhausted. `critical` is intentionally not
        consulted here — it is a label for alerting, not eviction
        protection, because real context-window truncation this naive
        does not know or care which segment was important."""
        kept: list[Segment] = []
        used = 0
        for segment in reversed(self.segments):
            if used + segment.token_count <= max_tokens:
                kept.append(segment)
                used += segment.token_count
        kept.reverse()
        self._rendered = kept
        return kept

    def to_messages(self) -> list[dict]:
        role_by_source = {"system_policy": "system"}
        return [{"role": role_by_source.get(s.source, "user"), "content": s.content} for s in self._rendered]
```

# sdk/context_mri/diff.py
from __future__ import annotations

from .models import LifecycleEvent, Segment

_trace_history: dict[str, tuple[str, list[Segment]]] = {}


def diff_against_previous(trace_id: str, call_id: str, rendered_segments: list[Segment]) -> list[LifecycleEvent]:
    events: list[LifecycleEvent] = []
    previous = _trace_history.get(trace_id)

    if previous is not None:
        previous_call_id, previous_segments = previous
        previous_ids = {s.segment_id for s in previous_segments}
        current_ids = {s.segment_id for s in rendered_segments}

        for missing_id in previous_ids - current_ids:
            events.append(LifecycleEvent(segment_id=missing_id, type="evicted", previous_call_id=previous_call_id))
        for new_id in current_ids - previous_ids:
            events.append(LifecycleEvent(segment_id=new_id, type="added", previous_call_id=previous_call_id))

    _trace_history[trace_id] = (call_id, rendered_segments)
    return events


def reset_trace_history() -> None:
    """Test-only: clears the module-level trace buffer between tests."""
    _trace_history.clear()
```

# sdk/context_mri/instrument.py
from __future__ import annotations

import os
from typing import Any, Callable

import requests

from .builder import ContextBuilder
from .diff import diff_against_previous
from .models import Segment

BACKEND_URL = os.getenv("CONTEXT_MRI_BACKEND_URL", "http://127.0.0.1:3000")


def _segment_to_dict(segment: Segment) -> dict:
    return {
        "segment_id": segment.segment_id,
        "source": segment.source,
        "critical": segment.critical,
        "token_count": segment.token_count,
        "present": True,
    }


def instrument_call(api_fn: Callable[..., Any]) -> Callable[..., Any]:
    """Wrap a model-calling function so every call is rendered, diffed
    against the previous call in the trace, and reported to the backend."""

    def wrapped(builder: ContextBuilder, max_tokens: int, timestamp: str, **kwargs: Any) -> Any:
        rendered = builder.render(max_tokens=max_tokens)
        events = diff_against_previous(builder.trace_id, builder.call_id, rendered)

        response = api_fn(builder.to_messages(), **kwargs)

        payload = {
            "trace_id": builder.trace_id,
            "call_id": builder.call_id,
            "timestamp": timestamp,
            "segments": [_segment_to_dict(s) for s in rendered],
            "events": [
                {"segment_id": e.segment_id, "type": e.type, "previous_call_id": e.previous_call_id}
                for e in events
            ],
            "token_budget": max_tokens,
            "token_used": sum(s.token_count for s in rendered),
            "model_response": response,
        }
        requests.post(f"{BACKEND_URL}/events", json=payload, timeout=5)
        return response

    return wrapped


def report_tool_call(call_id: str, tool_name: str, args: dict, result: Any) -> None:
    requests.post(
        f"{BACKEND_URL}/tool_calls",
        json={"call_id": call_id, "tool_name": tool_name, "args": args, "result": result},
        timeout=5,
    )

from typing import Any


def _parse_tool_call(response_json: dict) -> Any:
    for item in response_json.get("output", []):
        if item.get("type") == "function_call":
            return {
                "name": item["name"],
                "args": json.loads(item.get("arguments") or "{}"),
            }
    return None


def call_openai_messages(messages: list, model: str = "gpt-5.6", tools: list | None = None) -> Any:
    """Send a structured messages array to the OpenAI Responses API and
    return a normalized {"tool_call": ..., "raw": ...} response."""
    api_key = _required_env("OPENAI_API_KEY")

    body: dict[str, Any] = {"model": model, "input": messages}
    if tools:
        body["tools"] = tools

    request = Request(
        OPENAI_RESPONSES_URL,
        data=json.dumps(body).encode("utf-8"),
        headers={
            "Authorization": f"Bearer {api_key}",
            "Content-Type": "application/json",
        },
        method="POST",
    )

    with urlopen(request) as response:
        response_json = json.load(response)

    return {"tool_call": _parse_tool_call(response_json), "raw": response_json}

# sdk/demo_app/insurance_agent.py
from __future__ import annotations

import uuid
from datetime import datetime, timezone
from typing import Any, Callable

from context_mri.builder import ContextBuilder
from context_mri.instrument import instrument_call, report_tool_call

from .policy import POLICY_TEXT
from .retrieval import retrieve
from .tools import TOOL_DEFINITIONS, TOOL_IMPLEMENTATIONS


def _now_iso() -> str:
    return datetime.now(timezone.utc).isoformat()


def run_conversation(
    call_openai_messages: Callable[..., Any],
    claim_category: str,
    user_turns: list[str],
    max_tokens: int,
) -> list[dict]:
    trace_id = str(uuid.uuid4())
    history: list[str] = []
    results: list[dict] = []
    wrapped_call = instrument_call(call_openai_messages)

    for turn_index, user_message in enumerate(user_turns):
        call_id = f"{trace_id}-{turn_index}"
        builder = ContextBuilder(trace_id=trace_id, call_id=call_id)
        builder.add("system_policy", POLICY_TEXT, critical=True)
        builder.add("retrieved_docs", retrieve(claim_category))
        for prior_turn in history:
            builder.add("history", prior_turn)
        builder.add("user_input", user_message)

        response = wrapped_call(
            builder,
            max_tokens=max_tokens,
            timestamp=_now_iso(),
            tools=TOOL_DEFINITIONS,
        )

        tool_call = response.get("tool_call")
        if tool_call:
            tool_name = tool_call["name"]
            args = tool_call["args"]
            result = TOOL_IMPLEMENTATIONS[tool_name](**args)
            report_tool_call(call_id, tool_name, args, result)
            results.append({"call_id": call_id, "tool_name": tool_name, "result": result})

        history.append(user_message)

    return results
```

// dashboard/src/api.js
const API_BASE = import.meta.env.VITE_API_BASE || 'http://127.0.0.1:3000';

export async function fetchTrace(traceId) {
  const res = await fetch(`${API_BASE}/traces/${traceId}`);
  if (!res.ok) throw new Error(`Failed to fetch trace ${traceId}: ${res.status}`);
  return res.json();
}

export async function fetchAlerts(traceId) {
  const res = await fetch(`${API_BASE}/alerts?trace_id=${traceId}`);
  if (!res.ok) throw new Error(`Failed to fetch alerts for ${traceId}: ${res.status}`);
  return res.json();
}

// dashboard/src/components/DiffView.jsx
export function DiffView({ previousCall, currentCall }) {
  const evicted = currentCall.events.filter((e) => e.type === 'evicted');
  const added = currentCall.events.filter((e) => e.type === 'added');

  return (
    <div className="diff-view">
      <h3>
        Changes from {previousCall ? previousCall.call_id : '(start)'} → {currentCall.call_id}
      </h3>
      {evicted.length === 0 && added.length === 0 && <p>No context changes.</p>}
      {evicted.map((event) => (
        <p key={event.segment_id} className="diff-evicted">
          − removed segment {event.segment_id}
        </p>
      ))}
      {added.map((event) => (
        <p key={event.segment_id} className="diff-added">
          + added segment {event.segment_id}
        </p>
      ))}
    </div>
  );
}
